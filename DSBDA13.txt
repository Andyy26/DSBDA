object EvenOddChecker {
  def main(args: Array[String]): Unit = {
    println("Enter a number:")
    val input = scala.io.StdIn.readInt()

    if (input % 2 == 0)
      println(s"$input is Even")
    else
      println(s"$input is Odd")
  }
}
